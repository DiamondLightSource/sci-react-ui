export const Marker = ({ shape }) => (
  <g dangerouslySetInnerHTML={{ __html: markerSvg(shape) }} />

export const core = palette.thresholds.coreSlots;
export const total = palette.slots.length;

export const Axes = ({ W, H, P }) =>
  [0, 25, 50, 75, 100].map((v) => {
    const y = P.t + ((H - P.t - P.b) * (100 - v)) / 100;
    return (
      <g key={v}>
        <line x1={P.l} x2={W - P.r} y1={y} y2={y} stroke="var(--ds-border-subtle)" />
        <text x={P.l - 6} y={y + 3} textAnchor="end" fontSize="10" fill="var(--ds-on-surface-variant)">{v}</text>
      </g>
    );
  });

export const ChartMarker = ({ slot, x, y }) => (
  <g transform={`translate(${x} ${y})`} style={{ color: `var(--ds-chart-${slot.slot})` }}>
    <Marker shape={slot.marker} />
  </g>
);

export const LineChart = ({ count }) => {
  const W = 720, H = 240, P = { l: 30, r: 10, t: 10, b: 14 }, pts = lineSeries[0].length;
  const X = (i) => P.l + ((W - P.l - P.r) * i) / (pts - 1);
  const Y = (v) => P.t + (H - P.t - P.b) * (1 - v / 100);
  return (
    <svg viewBox={`0 0 ${W} ${H}`} role="img" aria-label={`Line chart with ${count} series`} style={{ width: "100%", minWidth: W, height: "auto", display: "block" }}>
      <Axes W={W} H={H} P={P} />
      {palette.slots.slice(0, count).map((s, k) => {
        const base = lineSeries[k];
        const d = base.map((y, i) => `${i ? "L" : "M"}${X(i).toFixed(1)} ${Y(y).toFixed(1)}`).join("");
        return (
          <g key={s.slot}>
            <path d={d} fill="none" stroke={`var(--ds-chart-${s.slot})`} strokeWidth="1.75" strokeLinejoin="round" />
            {[0, 1, 2, 3].map((q) => {
              const t = (0.04 + (0.92 * (q + (k + 0.5) / count)) / 4) * (pts - 1);
              const i0 = Math.min(pts - 2, Math.floor(t));
              const yv = base[i0] + (base[i0 + 1] - base[i0]) * (t - i0);
              return <ChartMarker key={q} slot={s} x={X(t)} y={Y(yv)} />;
            })}
          </g>
        );
      })}
    </svg>
  );
};

export const ScatterChart = ({ count }) => {
  const W = 360, H = 240, P = { l: 30, r: 10, t: 10, b: 14 };
  return (
    <svg viewBox={`0 0 ${W} ${H}`} role="img" aria-label={`Scatter chart with ${count} groups`} style={{ width: "100%", minWidth: W, height: "auto", display: "block" }}>
      <Axes W={W} H={H} P={P} />
      {palette.slots.slice(0, count).map((s, k) => {
        return (
          <g key={s.slot}>
            {scatterSeries[k].map(([fx, fy], i) => (
              <ChartMarker
                key={i}
                slot={s}
                x={P.l + (W - P.l - P.r) * fx}
                y={P.t + (H - P.t - P.b) * (1 - fy)}
              />
            ))}
          </g>
        );
      })}
    </svg>
  );
};

# Chart colours

<span class="lead">

  colours per mode, ordered so the first few are the easiest to tell apart.

Use these colours to tell data series apart. They are separate from the UI colours: do not use intent colours as a chart palette, and do not use chart colours to signal status. See **[Colour](/docs/foundations-colour--docs)** for UI colour and **[Theme Colours](/docs/foundations-theme-colours--docs)** for the semantic roles.

## Using the palette

<div className="ds-note">

### Use series in order

Start at `--ds-chart-1` and take the next slot for each new series. Slots 1 to {core} stay separable under colour-vision deficiency. Slots {core + 1} to {total} have weaker separation for some pairs, so use them only when you need more than {core} series. Skipping or reordering slots can put weaker pairs together.

### Draw charts on the surface or the background

The colours are checked for {palette.thresholds.contrast}:1 contrast on `--ds-surface` and `--ds-background`. On raised surfaces, tooltips and dialogs they are only checked to {palette.thresholds.contrastFloor}:1, so label series directly or add a data table there.

### Pair every colour with a marker

Colour alone is not enough. Every slot has its own marker shape. Draw lines at least {palette.secondChannel.minLineWidth}px wide and markers at least {palette.secondChannel.minMarkerSize}px across, with a {palette.secondChannel.markerHalo}px `--ds-surface` halo on filled markers. Markers matter most in light mode, where several colours have similar lightness, and for greyscale, print and forced colours.

### Keep series colours for marks

Do not colour labels, values or legend text with a series colour. Set text with the text tokens and show the series colour as a swatch.

### Regions on images

Draw a region with a solid series-coloured border over a soft fill: `rgb(var(--ds-chart-1-channel) / var(--ds-chart-fill-region))`. Hover raises the fill to `--ds-chart-fill-region-hover`. Selected raises it again to `--ds-chart-fill-region-selected` and thickens the border from 2px to 3px, so selection does not depend on colour alone. Add a soft halo outside the border, `--ds-surface` at about 55% opacity and 1px wide, so the border reads over bright and dark pixels. Keyboard focus uses the shared focus ring, not a chart colour. The fills are decorative, and the border's contrast depends on the image behind it, so check it on your own images.

### Labels next to a line

Use a tint of the series colour for the label background, `rgb(var(--ds-chart-1-channel) / var(--ds-chart-fill-label))`, with a 1px border in the series colour and `--ds-on-surface` text. Hover uses `--ds-chart-fill-label-hover`. Selected uses `--ds-chart-fill-label-selected`, a 2px border and bolder text. Keep labels on `--ds-surface` or `--ds-background`, where the text keeps at least {palette.thresholds.textContrast}:1 on every tint in every state. Never use the series colour for the text itself.

### Never use them for status

{palette.nearStatus.length} of the {palette.slots.length} series colours ({[...palette.nearStatus].sort((a, b) => a.slot - b.slot).map((n) => `${n.slot} ${n.name}`).join(", ")}) sit close to a status or brand colour, {palette.nearStatus[0].name} most of all (<code>{palette.nearStatus[0].token}</code>). Use the intent colours for meaning, and the chart colours only to tell series apart.

## Accessibility

- Series colours are graphical objects, so they need {palette.thresholds.contrast}:1 contrast against the surface they sit on (WCAG 1.4.11). They are checked on `--ds-surface` and `--ds-background`.
- Colour must not be the only way to tell series apart (WCAG 1.4.1). Add a legend, markers and direct labels or a data table.
- Overlapping translucent fills blend into colours outside the palette. Prefer lines, or keep overlaps to two series. Separate adjacent filled segments, such as stacked bars, with a 1px to 2px `--ds-surface` gap.
- In forced-colours mode all series can collapse to one system colour. Keep `forced-color-adjust: none` on series marks, or rely on markers.

## Maintaining the palette

The tokens are generated from `tools/chart-palette/palette.mjs`. Do not edit `src/styles/diamondDS/DiamondDSChartTokens.css` by hand. After changing the palette, run `node tools/chart-palette/build.mjs`, which rewrites the tokens and the generated report and preview in `tools/chart-palette/`. The report lists the current contrast and separation figures, and a test fails if the outputs are stale or a check fails.
